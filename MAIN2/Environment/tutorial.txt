# OBSERVATION EXPERIMENT
#
# The instruction the ATCO is evaluating is observed as its TARGET heading and speed
# (pend_hdg, pend_spd in the same frame as h_cmd and v_cmd) instead of the offset from the
# commanded heading and speed. New aircraft may also spawn in a conflict predicted within
# t_warn (--spawn-conflicts); they must still be 15 NM from all traffic, and d_chord still
# applies. This gives about 60% more conflicts per flight hour.
#
# Run everything from the MAIN2 folder:
#     cd <wherever MAIN2 is>
# Check you are in the right place -- this must list Environment, Training, Validation:
#     ls
# The models are written to MAIN2/experiment Gijs.

# terminal 1: none
python -m Training.train --delay none --seed 1 --pending-obs target --spawn-conflicts --timesteps 300000000 --runs-root "experiment Gijs"

# terminal 2: lognormal 30 s
python -m Training.train --delay lognormal --delay-mean 30 --seed 1 --pending-obs target --spawn-conflicts --timesteps 300000000 --runs-root "experiment Gijs"

# terminal 3: deterministic 30 s
python -m Training.train --delay deterministic --delay-mean 30 --seed 1 --pending-obs target --spawn-conflicts --timesteps 300000000 --runs-root "experiment Gijs"

# tensorboard
python -m tensorboard.main --logdir "experiment Gijs" --port 6016


================================================================================
RESUMING AFTER A DISCONNECT
================================================================================

# Same commands, plus --resume. Each run continues from its own last checkpoint,
# keeping the step count and the TensorBoard curves continuous. At most
# --save-every steps (500,000) are lost per run.
#
# Do NOT delete "experiment Gijs". Do NOT pass --overwrite; that starts from scratch.

# terminal 1: none
python -m Training.train --delay none --seed 1 --pending-obs target --spawn-conflicts --timesteps 300000000 --runs-root "experiment Gijs" --resume

# terminal 2: lognormal 30 s
python -m Training.train --delay lognormal --delay-mean 30 --seed 1 --pending-obs target --spawn-conflicts --timesteps 300000000 --runs-root "experiment Gijs" --resume

# terminal 3: deterministic 30 s
python -m Training.train --delay deterministic --delay-mean 30 --seed 1 --pending-obs target --spawn-conflicts --timesteps 300000000 --runs-root "experiment Gijs" --resume
