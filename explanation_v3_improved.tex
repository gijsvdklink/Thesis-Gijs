\section{Overview}
% ─────────────────────────────────────────────────────────────────────────────

Single upper-airspace sector, A320 aircraft at FL350 ($M=0.78$, $\approx450$\,kt TAS).
Per episode: $N \sim \mathcal{U}\{2,\ldots,15\}$ aircraft, density $\sim\mathcal{U}(5\,000,\,15\,000)$\,km$^2$/aircraft, area $= N\times$density.
Sector boundary: random convex polygon, 5--7 vertices, isoperimetric ratio $\geq 0.65$.
Minimum spawn separation 10\,NM; exiting aircraft replaced immediately.
Episode length $\approx60$\,min (2.5 sector crossings).
Each RL step spans 5\,s (10 BlueSky sub-steps of 0.5\,s); a $30^\circ$ turn takes $\approx26.5$\,s ($\approx5$--6 steps), speed changes 2--5 steps.

% ─────────────────────────────────────────────────────────────────────────────
\section{Environment Loop}
\label{sec:loop}
% ─────────────────────────────────────────────────────────────────────────────

\begin{algorithm}[h]
\caption{\textsc{Reset}}
\label{alg:reset}
\begin{algorithmic}[1]
\State $N \sim \mathcal{U}\{2,\ldots,15\}$;\quad density $\sim\mathcal{U}(5\,000,\,15\,000)$;\quad area $\leftarrow N\times$density
\State Generate convex polygon;\quad compute $T_\text{max}$ from diameter and cruise speed
\For{slot $s = 1,\ldots,N$} place aircraft on perimeter arc $s$ with jitter \EndFor
\State Compute $\mathbf{U}$;\quad $f \leftarrow \arg\max_i \max_{j\neq i} u_{ij}$;\quad \Return $\mathbf{o}(f)$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[h]
\caption{\textsc{Step}(action)}
\label{alg:step}
\begin{algorithmic}[1]
\State Spawn pending replacements
\If{heading offset $\delta$} $\psi_\text{cmd} \leftarrow \psi_\text{dest}(f)+\delta$
\ElsIf{Mach $\Delta M$} $M_\text{cmd} \leftarrow \operatorname{clip}(M_\text{cmd}+\Delta M,\,0.70,\,0.82)$
\EndIf
\For{$k=1,\ldots,10$} advance BlueSky sub-step \EndFor
\State $\ell_\text{LoS} \leftarrow \mathbf{1}[\exists\,j: d(f,j)<d_\text{sep}]$
\State Process exits;\quad recompute $\mathbf{U}$;\quad select $f'$ (Section~\ref{sec:urgency})
\State $r \leftarrow r_\text{conflict}(f)+r_\text{LoS}(f)+r_\text{nav}(f)-w_\text{work}\cdot c_f$
\State \Return $\mathbf{o}(f'),\;r,\;\text{truncated}=(\text{step}\geq T_\text{max})$
\end{algorithmic}
\end{algorithm}

% ─────────────────────────────────────────────────────────────────────────────
\section{Urgency Matrix and Focus Selection}
\label{sec:urgency}
% ─────────────────────────────────────────────────────────────────────────────

\begin{equation}
    u_{ij} =
    \begin{cases}
        1 + 9\!\left(1 - \dfrac{d_{ij}}{d_\text{sep}}\right)
            & d_{ij} < d_\text{sep}
              \quad\text{(LoS,\;} u_{ij}\in[1,10]\text{)} \\[10pt]
        \dfrac{T_\text{warn} - t_\text{CPA}^{ij}}{T_\text{warn}}
            & d_\text{CPA}^{ij} < d_\text{sep},\;
              0 \leq t_\text{CPA}^{ij} \leq T_\text{look} \\[8pt]
        0   & \text{otherwise}
    \end{cases}
    \label{eq:urgency}
\end{equation}

$d_\text{sep}=5$\,NM,\quad $T_\text{warn}=600$\,s,\quad $T_\text{look}=900$\,s.

Focus: $f = \arg\max_i \max_{j\neq i} u_{ij}$.
No-conflict fallback: aircraft with largest heading deviation $(1-\cos\Delta\psi)/2$,
subject to a switch margin of 0.05 to prevent oscillation.

\paragraph{Commitment rule.}
Focus locked while $u_f>0$ or cooldown $< n_\text{clear}=5$ steps.
Emergency override: another aircraft reaching $u_\text{emerg}=0.8$
($t_\text{CPA}\approx120$\,s) releases the lock immediately.

% ─────────────────────────────────────────────────────────────────────────────
\section{Action Space}
% ─────────────────────────────────────────────────────────────────────────────

10 discrete actions (6 heading, 4 speed):
\begin{align}
    \psi_\text{cmd} &= \psi_\text{dest} + \delta,\qquad
    \delta \in \{-30^\circ,\,-15^\circ,\,0^\circ,\,+15^\circ,\,+30^\circ,\,\varnothing\} \\
    M_\text{cmd} &\leftarrow \operatorname{clip}(M_\text{cmd}+\Delta M,\,0.70,\,0.82),\qquad
    \Delta M \in \{-0.04,\,-0.02,\,+0.02,\,+0.04\}
\end{align}
Direct ($0^\circ$) and hold ($\varnothing$) are free ($c_f=0$); turns $c_f=1.0$; speed $c_f=0.5$.

% ─────────────────────────────────────────────────────────────────────────────
\section{Observation Space ($\mathbb{R}^{25}$)}
% ─────────────────────────────────────────────────────────────────────────────

Ego-centric from focus aircraft. Intruder slots sorted by urgency $\downarrow$
then CPA distance $\uparrow$; diverging pairs sentinel-sorted to back.
Empty/diverging slot: $(1,1,0,1,1)$.

Intruder positions in ownship heading frame, normalised by
$d_\text{warn} = T_\text{warn}\cdot v_\text{nom} = 75$\,NM (10-min action horizon).
CPA offset:
\begin{equation}
    \begin{pmatrix}\Delta x^\text{CPA}\\\Delta y^\text{CPA}\end{pmatrix}
        = \begin{pmatrix}\Delta x\\\Delta y\end{pmatrix}
        + t_\text{CPA}\begin{pmatrix}\Delta v_e\\\Delta v_n\end{pmatrix}.
\end{equation}

% ─────────────────────────────────────────────────────────────────────────────
\section{Reward Function}
% ─────────────────────────────────────────────────────────────────────────────

\begin{equation}
    r_t = r_\text{conflict} + r_\text{LoS} + r_\text{nav} - w_\text{work}\cdot c_f
\end{equation}

Let $\sigma_t = \mathbf{1}[\max_{j\neq f} u_{fj} > 0]$:
\begin{equation}
    r_\text{conflict} =
    \begin{cases}
        +w_\text{resolve} & \sigma_{t-1}=1,\;\sigma_t=0 \quad\text{(resolved)}\\
        -w_\text{resolve} & \sigma_{t-1}=0,\;\sigma_t=1 \quad\text{(new)}\\
        -w_\text{resolve} & \sigma_{t-1}=1,\;\sigma_t=1 \quad\text{(ongoing)}\\
        0 & \text{otherwise}
    \end{cases}
\end{equation}

On the first step after a focus switch: $r_\text{conflict}=0$ (grace step) and
$\sigma_{t-1}$ is synchronised to the aircraft's actual state.

\begin{equation}
    r_\text{LoS} = -w_\text{LoS}\cdot\mathbf{1}[\ell_\text{LoS}]
    \qquad
    r_\text{nav} = w_\text{nav}\cdot\bigl(\cos\Delta\psi_\text{act}^{\,t}
                                          - \cos\Delta\psi_\text{act}^{\,t-1}\bigr)
\end{equation}

\section{Training}
% ─────────────────────────────────────────────────────────────────────────────

PPO (Stable-Baselines3), 4 parallel environments, two-layer MLP.
Observations not normalised (\texttt{norm\_obs=False}): all features are
bounded by design.
