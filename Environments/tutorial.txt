================================================================================
ACTION-RESPONSE DELAY EXPERIMENT -- COMMANDS
================================================================================

--------------------------------------------------------------------------------
CHECKS BEFORE TRAINING
--------------------------------------------------------------------------------

    python -m Environments.v4.test_delays

--------------------------------------------------------------------------------
TRAINING -- 10 terminals
--------------------------------------------------------------------------------

    # terminal 1
    python -m Training.v4_train --delay none --seed 0 --n-envs 2

    # terminals 2-4
    python -m Training.v4_train --delay deterministic --delay-mean 15 --seed 0 --n-envs 2
    python -m Training.v4_train --delay lognormal     --delay-mean 15 --seed 0 --n-envs 2
    python -m Training.v4_train --delay probabilistic --delay-mean 15 --seed 0 --n-envs 2

    # terminals 5-7
    python -m Training.v4_train --delay deterministic --delay-mean 30 --seed 0 --n-envs 2
    python -m Training.v4_train --delay lognormal     --delay-mean 30 --seed 0 --n-envs 2
    python -m Training.v4_train --delay probabilistic --delay-mean 30 --seed 0 --n-envs 2

    # terminals 8-10
    python -m Training.v4_train --delay deterministic --delay-mean 45 --seed 0 --n-envs 2
    python -m Training.v4_train --delay lognormal     --delay-mean 45 --seed 0 --n-envs 2
    python -m Training.v4_train --delay probabilistic --delay-mean 45 --seed 0 --n-envs 2

--------------------------------------------------------------------------------
TENSORBOARD
--------------------------------------------------------------------------------

    tensorboard --logdir Runs_saved/experiments --port 6006

    # output directories
    #   none/   deterministic_15s/   lognormal_15s/   probabilistic_15s/
    #           deterministic_30s/   lognormal_30s/   probabilistic_30s/
    #           deterministic_45s/   lognormal_45s/   probabilistic_45s/

--------------------------------------------------------------------------------
VALIDATION -- 23 terminals
--------------------------------------------------------------------------------

    # the full set of commands lives in Validation/experiments.txt

    # CSVs go to Validation/report_results/results, figures to .../figures.
    # Both folders are created automatically, and a bare file name lands in the
    # right one, so no paths are needed on the command line.

    # one policy in one delay world
    python -m Validation.mc_evaluate --model Runs_saved/experiments/none/*/last_model.zip \
        --episodes 200 --seed 1 --delay lognormal --delay-mean 30 --out run.csv

    # no CR at all, for comparison: leave out --model
    python -m Validation.mc_evaluate --episodes 200 --seed 1 --out no_cr.csv

    # every condition flies the same episodes: --seed picks where in the held-out
    # TEST_SEEDS pool the run starts, so the same --seed gives the same scenarios

    # means with 95% bootstrap intervals, printed as a table
    python -m Validation.summarise policy=run.csv "no CR=no_cr.csv"

    # the two report figures
    python -m Validation.plot_report

--------------------------------------------------------------------------------
VISUALISATION
--------------------------------------------------------------------------------

    # trained policy
    python visualisation/visualise.py --model Runs_saved/experiments/none/*/last_model.zip \
        --vecnorm Runs_saved/experiments/none/*/last_model_vecnorm.pkl \
        --n_ac 15 --density 0.0001 --delay lognormal --delay-mean 45

    # do-nothing baseline
    python visualisation/visualise.py --hold --n_ac 15 --density 0.0001

    # find a seed where two aircraft actually come into conflict
    python visualisation/find_conflict_seed.py --seeds 40

    # single aircraft, one advisory then silence, delay countdown on screen
    python visualisation/visualise.py --n_ac 1 --density 0.0001 \
        --delay deterministic --delay-mean 45 --fixed-action 2 --seq-once \
        --seed 9 --steps 95 --fps 2

    # single aircraft turning left 60 every step
    python visualisation/visualise.py --n_ac 1 --density 0.0001 \
        --delay deterministic --delay-mean 45 --fixed-action 0 --seed 9 --fps 2

    # render to mp4 instead of a window
    python visualisation/visualise.py --n_ac 1 --density 0.0001 \
        --delay deterministic --delay-mean 45 --fixed-action 0 --seed 9 --fps 2 \
        --steps 95 --mp4 solo_L60_det45.mp4
